\section{Variance of the Perturbed Estimator}

\begin{proposition}
    For any perturbation $\Delta = (\Delta_1, \dots, \Delta_d)$,
    \begin{align}
        \Var(\hat{g}[c]) = \underbrace{\vphantom{\sum_{j,k}}V_{Q}}_{\text{variance at }c=Q} - 2\underbrace{\sum_{j=1}^d\sum_{k=1}^d
     \mathbb{E}\big[m_k(b_{-k})^{\top}u_j\big]}_{=:B^{\top}\Delta\ (\text{cross term})} + \underbrace{\sum_{j=1}^d\sum_{j'=1}^d
     \mathbb{E}\big[u_j^{\top}u_{j'}\big]}_{=:\Delta^{\top}C\Delta\ (\text{quadratic term})}
    \end{align}
\end{proposition}

\begin{proof}
    The variance can thus be computed as:
    \begin{align}
        \Var(\hat{g}[c]) &= \E_{p(b|\theta)} [\|\hat{g}[c] - g\|^2] = \E[\|\hat{g}[Q] - g - U\|^2]\\
        &=\E[\|\hat{g}[Q] - g\|^2] - 2\E[(\hat{g}[Q] - g)^\top U] + \E[U^\top U]
    \end{align}
    For the second term, since $\E[u_j] = \E[\E[u_j | b_{-j}]] = 0$, thus $g^\top \E[U]=0$. 
\end{proof}

\begin{example}
    Consider two independent Bernoulli variables $b_1, b_2 \sim \mathrm{Bernoulli}(p)$ with a shared parameter $p = \sigma(\theta)$. Let $f(b_1, b_2) = b_1b_2$. At $\theta = 0$, we have $p=1/2$ and $\nabla_\theta p = 1/4$, so 
    \begin{align}
        \nabla_\theta \gL(\theta) =  \nabla_\theta \E_{p(\theta)}[f]=\frac14.
    \end{align}
    For the condition-mean control variates,
    \begin{align}
        Q_1 (b_2, v) = vb_2 \quad Q_2 (b_1, v) = b_1 v,
    \end{align}
    and hence $Q_j(b_{-j},b_j)=f(b)$.

    Intuitively, $Q_1$ averages over all possible values of $b_1$, so the randomness from $b_1$ is removed; however, $b_2$ is still sampled and remains random. Similiarly, $Q_2$ removes the randomness from $b_2$ but not from $b_1$. Thus, substituting $c=Q$ gives 
    \begin{align}
        \hat g[Q] = \frac{1}{4} b_2 + \frac{1}{4}b_1 = \frac{1}{4} (b_1 + b_2),
    \end{align}
    and thus
    \begin{align}
        \Var(\hat{g}[Q]) = \frac{1}{32} > 0.
    \end{align}

    Now consider $c_1 = Q_1 + 1/4$, $c_2 = Q_2 + 1/4$. Since 
    $\nabla_\theta \log p(b_j)=b_j-\frac12$, the additional terms exactly cancel the remaining randomness:
    \begin{align}
        \hat g[c]
=
\frac14(b_1+b_2)
-\frac14\left[
\left(b_1-\frac12\right)
+
\left(b_2-\frac12\right)
\right]
=
\frac14.
    \end{align}
    Hence,
    \begin{align}
        \Var \hat g[c] = 0 < \Var(\hat{g}[Q]).
    \end{align}
\end{example}

\subsection{Discussion of Our Estimator}

\begin{itemize}
    \item When can we reduce the variance?
    Since the diagonal terms $\E[m(b_{-j})^\top u_j] = 0$, we need the cross terms to be non-zero.
    \item Q: Can our estimator minimize the variance?

\begin{theorem}
    Write the variance as $\Var[\hat{g}] = V_Q - 2B^\top \Delta + \Delta^\top C\Delta$, then the minimizes are:
    \begin{align}
        c^* = Q + \Delta^*, \quad \Delta^* = C^\dagger B, \quad \Var(\hat{g}[c^*]) = V_Q - B^\top C^\dagger B.
    \end{align}
\end{theorem}
\end{itemize}

\subsection{Running Example: the MCP Coding-Coach Task}
\label{sec:mcp}

A trainable model $G_\theta$ (Llama-3.2-3B-Instruct with a LoRA adapter, $\theta\in\mathbb{R}^P$,
$P=917{,}504$) is shown a programming problem $x$ and acts as a \emph{coding coach who also
configures the engineer's workspace}. It samples one line of JSON,
\begin{verbatim}
{"signature": "merge_intervals(intervals)",
 "plan": "Sort by start, then fold overlapping ranges.",
 "edge_cases": ["empty list", "touching intervals"],
 "tools": ["run_python", "edge_case_checklist"],
 "tool_notes": {"run_python": "try your code on examples you invent"}}
\end{verbatim}
This JSON \emph{is} the sampled object: $b=y_{1:T}$, $T\approx172$ tokens. A black box $O$ --- a
separate LLM behind an API --- then sees \emph{only} this JSON, never the problem statement, plus the
MCP tools the coach chose to expose, and writes code $z$. Finally $f(b)\in[0,1]$ is the fraction of
hidden tests that $z$ \emph{fails}. One $O$-call per rollout; no gradient passes through $O$.

Two features of the token sequence drive everything below. Most tokens are structural
(\texttt{\{}, \texttt{"signature"}, \texttt{:}, \texttt{,}) and $p_t$ is nearly a point mass there.
A minority are genuine decisions: which tools to list, whether the plan says ``sort first'', which
edge cases to name. And $Q_t(v)=\E[f\mid y_{<t},y_t=v]$ is the expected test-failure rate if the
coach's next token is $v$ --- not computable, so we fit an approximation and correct it with $\Delta$.

\subsection{Autoregressive Instantiation: Exact Computation of $B$ and $C$}

Let $b=y_{1:T}$ with $p(b\mid\theta)=\prod_t p_t(y_t\mid y_{<t})$, vocabulary $\mathcal{V}$. Restrict
$c_t$ to the prefix, $c_t=c_t(y_{<t},v)$. Write $s_t:=\nabla_\theta\log p_t(y_t\mid y_{<t})$ and
$S:=[s_1,\dots,s_T]^\top\in\mathbb{R}^{T\times P}$. Then
\begin{align}
\hat g[c]=\sum_{t=1}^T\big(f(b)-c_t(y_t)\big)s_t
\;+\;\nabla_\theta\sum_{t=1}^T\sum_{v\in\mathcal{V}}p_t(v)\,c_t(v).
\label{eq:ghat-ar}
\end{align}
The second term is the price of letting $c_t$ depend on $v$; it vanishes identically when $c_t$ does not.

Fix a dictionary $\delta^{(1)},\dots,\delta^{(K)}$ and set $\Delta_t(v)=\sum_k w_k\,\delta^{(k)}_t(v)$.
Since \eqref{eq:ghat-ar} is affine in $c$, we have $U=\sum_k w_k u_k$ with
\begin{align}
u_k=-\sum_t\delta^{(k)}_t(y_t)\,s_t+\nabla_\theta\sum_t\sum_v p_t(v)\,\delta^{(k)}_t(v),
\label{eq:uk}
\end{align}
so the Theorem applies verbatim with $\Delta\equiv w\in\mathbb{R}^K$, $B_k=-\E[u_k^\top\hat g[Q]]$ and
$C_{kl}=\E[u_k^\top u_l]$.

\paragraph{The obstruction, concretely.} Suppose we want to test the candidate ``\emph{while the
\texttt{plan} field is being written, subtract a multiple of how surprising each candidate token is}'',
i.e.\ $\delta_t(v)=\varphi_3(t)\log p_t(v)$. Evaluating it naively means materialising the full
$T\times|\mathcal{V}|$ table --- $172\times128{,}256$ --- pushing it through \eqref{eq:ghat-ar} and taking
a backward pass, \emph{per rollout}. Over $768$ rollouts that is roughly $40$ minutes, for \emph{one}
candidate. Searching over dictionaries this way is hopeless.

The next Lemma says the table is never needed.

\begin{lemma}[$(c,d)$ representation]\label{lem:cd}
Let $\varphi_1,\dots,\varphi_J$ be prefix-measurable position weights and set
\begin{align}
\tau^{(1)}_t=\sum_v p_t(v)Q_t(v),\qquad
\tau^{(2)}_t=-\sum_v p_t(v)\log p_t(v),\qquad
\tau^{(3)}_t=\sum_v p_t(v)^2 .
\end{align}
Define $Z\in\mathbb{R}^{q\times P}$, $q=3J$, with rows $z_{mj}=\nabla_\theta\sum_t\varphi_j(t)\,\tau^{(m)}_t$.
Then every $u_k$, and $\hat g[Q]$ itself, takes the single form
\begin{align}
u=-S^\top c+Z^\top d,\qquad c\in\mathbb{R}^T,\ d\in\mathbb{R}^{q},
\end{align}
with $(c,d)$ available in closed form:
\begin{center}
\begin{tabular}{lll}
\hline
$\delta_t(v)$ & $c_t$ & $d$\\
\hline
$\psi(t)$ \quad ($v$-independent) & $\psi(t)$ & $0$\\
$\varphi_j(t)\,Q_t(v)$ & $\varphi_j(t)\,Q_t(y_t)$ & $+e_{1j}$\\
$\varphi_j(t)\,\log p_t(v)$ & $\varphi_j(t)\,\log p_t(y_t)$ & $-e_{2j}$\\
$\varphi_j(t)\,p_t(v)$ & $\varphi_j(t)\,p_t(y_t)$ & $+\tfrac12 e_{3j}$\\
\hline
$\hat g[Q]$ & $-a_t,\quad a_t:=f-Q_t(y_t)$ & $+e_{11}$\\
\hline
\end{tabular}
\end{center}
\end{lemma}

\begin{proof}
$c_t(\cdot)$ is detached, so $\nabla_\theta\sum_v p_t(v)c_t(v)=\sum_v\big(\nabla_\theta p_t(v)\big)c_t(v)$.
For $c_t=\psi(t)$ this is $\psi(t)\nabla_\theta\!\sum_v p_t(v)=\psi(t)\nabla_\theta 1=0$.
For $c_t=Q_t$ it is $\nabla_\theta\tau^{(1)}_t$.
For $c_t=\log p_t$, using $\nabla_\theta\sum_v p_t(v)=0$,
$\sum_v(\nabla p_t)\log p_t=\nabla_\theta\sum_v p_t\log p_t=-\nabla_\theta\tau^{(2)}_t$.
For $c_t=p_t$, $\sum_v(\nabla p_t)p_t=\tfrac12\nabla_\theta\sum_v p_t^2=\tfrac12\nabla_\theta\tau^{(3)}_t$.
\end{proof}

\paragraph{Reading Lemma~\ref{lem:cd}.} It splits \eqref{eq:uk} into two halves and shows each one sees
the candidate $\delta$ through very few numbers.

\emph{The score half} sees $\delta$ only through the $T$ values $\delta_t(y_t)$ --- what the candidate
happens to evaluate to \emph{at the token the coach actually emitted}. The other $|\mathcal{V}|-1$
entries of each row never appear. For our candidate $\varphi_3(t)\log p_t(v)$ this is
$c_t=\varphi_3(t)\log p_t(y_t)$: one number per token, already sitting in the rollout log.

\emph{The correction half} sees $\delta$ only through $d\in\mathbb{R}^{q}$, $q=21$, which merely
\emph{selects} among $21$ gradients computed once and for all. Our candidate has $d=-e_{23}$: it picks
out $\nabla_\theta\sum_t\varphi_3(t)H_t$, the gradient of the entropy summed over the \texttt{plan}
region. Nothing of size $T\times|\mathcal{V}|$ is ever built.

And if the candidate does not depend on $v$ at all --- ``subtract a constant over the first $8$
tokens'', ``subtract the fitted value of the prefix'' --- then $d=0$ outright, because
$\sum_v p_t(v)\psi(t)=\psi(t)$ no longer depends on $\theta$. Such candidates cost \emph{nothing}
beyond a re-weighting of scores already in hand.

\begin{proposition}[Sufficient statistic]\label{prop:suff}
Let $\Sigma_{SS}:=SS^\top\in\mathbb{R}^{T\times T}$, $\Sigma_{SZ}:=SZ^\top\in\mathbb{R}^{T\times q}$,
$\Sigma_{ZZ}:=ZZ^\top\in\mathbb{R}^{q\times q}$. For $u=-S^\top c+Z^\top d$ and $u'=-S^\top c'+Z^\top d'$,
\begin{align}
u^\top u'=c^\top\Sigma_{SS}\,c'-c^\top\Sigma_{SZ}\,d'-d^\top\Sigma_{SZ}^\top c'+d^\top\Sigma_{ZZ}\,d'.
\end{align}
Hence the per-sample triple $(\Sigma_{SS},\Sigma_{SZ},\Sigma_{ZZ})$ determines $B$ and $C$ for
\emph{every} dictionary of the above form --- exactly, with no further backward passes.
\end{proposition}

\paragraph{Reading Proposition~\ref{prop:suff}.} Entry $(\Sigma_{SS})_{tu}=s_t^\top s_u$ answers: how
aligned is the LoRA update produced by ``the \texttt{\{} at position $1$'' with the one produced by
``the word \texttt{sort} at position $60$''? That $173\times173$ matrix, plus two small blocks
involving $Z$, is \emph{everything} any dictionary can ever see about a rollout. So we compute it once
--- $130$\,kB per rollout, $16$ minutes across six GPUs --- and afterwards every candidate dictionary
we care to write down is a $K\times K$ linear solve taking milliseconds, exactly, with no further
passes through the model. The $40$-minute cost above is paid once in total rather than once per idea.

Storage is $T^2+Tq+q^2\approx3.4\times10^4$ numbers versus $P\approx9.2\times10^5$ for a single $u_k$.
Note $\Sigma_{SS}$ alone suffices only when $d\equiv0$, i.e.\ for $v$-independent $\Delta$.

\subsection{Scope: What Transfers, and What Is Specific to This Experiment}
\label{sec:scope}

\paragraph{Universal.} These hold for any $\theta$-parameterised distribution over discrete sequences,
whatever $f$ or $O$ is.
\begin{itemize}
\item The estimator \eqref{eq:ghat-ar}, and the fact that the vocabulary correction is the exact price
of $v$-dependence.
\item $\Var=V_Q-2B^\top w+w^\top Cw$ with $w^*=C^\dagger B$, for $\Delta$ linear in any dictionary.
This is just linearity of $\hat g$ in $c$.
\item The score half of $u_k$ is $-S^\top c$ with $c_t=\delta_t(y_t)$: only the emitted token's value
of $\delta$ ever enters.
\item $d=0$ if and only if $\delta$ is $v$-independent.
\item The bilinear identity of Proposition~\ref{prop:suff}, given the $(c,d)$ form.
\end{itemize}

\paragraph{Design choices --- re-derivable, but not automatic.}
\begin{itemize}
\item The three $v$-dependent families $\{Q_t,\log p_t,p_t\}$ and the constants $+1,-1,+\tfrac12$.
A different family set gives a different $Z$ and a different $q$; the derivation is mechanical, but
each family must be worked out and stored.
\item $J=7$ position buckets ($q=3J=21$) and the $90$-column $v$-independent dictionary $\psi$.
\item $Q_t(v)=\langle Ah_t,Be_v\rangle$.
\item \emph{Whether the compression pays at all}: it requires $T^2+Tq+q^2\ll P$. Here
$3.4\times10^4\ll9.2\times10^5$. For very long horizons, or a small $\theta$, storing $u_k$ directly is
cheaper and the Proposition is pointless.
\end{itemize}

\paragraph{Empirical --- this policy, this task.} Every number in the next subsection. The one most
likely to reverse elsewhere is the finding that $v$-dependence is nearly worthless. It follows from
$57\%$ of tokens having $\max_v p_t(v)>0.9$, which is what an instruction-tuned model emitting rigid
JSON looks like. Under a hotter temperature, an earlier checkpoint, a weaker base model, or a task
where the policy is genuinely torn at its decision points, $\Var_{v\sim p_t}(Q_t(v))$ is larger and the
$v$-dependent columns should earn more. Nothing in the method assumes the quantity is small: the
machinery \emph{prices} it, and here the price came out near zero.

\subsection{Experiments}

\paragraph{Setup.} The task of Section~\ref{sec:mcp}: Llama-3.2-3B-Instruct with LoRA $r{=}8$ on the
query and value projections ($P=917{,}504$), $n=768$ rollouts over $6$ programming problems,
$\bar T=172$ ($\max 224$), $J=7$ so $q=21$, $\E[f]=0.287$, $\Var(f)=0.177$. All nuisances ($Q$, the
value probe, the per-problem baseline, $w$, and the ridge $\lambda$) are fitted $12$-fold
leave-one-group-out; every number below is out-of-fold. Ratios are $\Var(\hat g[Q+\Delta^*])/\Var(\hat g[Q])$,
so smaller is better.

\paragraph{Main result.} $K=111$ ($90$ $v$-independent $+\;21$ $v$-dependent).
\begin{center}
\begin{tabular}{lccc}
\hline
estimator & $\Var$ & vs.\ REINFORCE & vs.\ $\hat g[Q]$\\
\hline
REINFORCE, per-problem baseline & $1732.0$ & $1.000$ & ---\\
$\hat g[Q]$ & $1647.3$ & $0.951$ & $1.000$\\
$\hat g[Q+\Delta^*]$ & $1207.5$ & $0.697$ & $\mathbf{0.733}$\\
\hline
\end{tabular}
\end{center}
Bootstrap $95\%$ CI $[0.680,0.787]$; $0.749$ on an independent LoRA draw with the configuration frozen.
Measured end-to-end (real $(T,|\mathcal{V}|)$ table, fresh backward) $0.7330$ against
Proposition~\ref{prop:suff}'s prediction $0.7328$.

\paragraph{The off-diagonal of $\Sigma_{SS}$ is not negligible.}
Although $a^\top\mathrm{diag}(\Sigma_{SS})a\,/\,a^\top\Sigma_{SS}a=0.936$, that measures the objective
\emph{value} at $\Delta=0$, not the \emph{argmin}:
\begin{center}
\begin{tabular}{lc}
\hline
$w$ fitted on & ratio (scored on full $\Sigma_{SS}$)\\
\hline
full $\Sigma_{SS}$ & $0.7328$\\
$\mathrm{diag}(\Sigma_{SS})$ & $0.8459$\\
\hline
\end{tabular}
\end{center}
with $\cos(w_{\text{full}},w_{\text{diag}})=0.18$. The token-wise weighted-regression surrogate
$\min_m\sum_t\|s_t\|^2(a_t-m_t)^2$ is therefore \emph{not} a safe substitute.

\paragraph{Value of $v$-dependence in $\Delta$.}
\begin{center}
\begin{tabular}{lc}
\hline
columns used & ratio\\
\hline
$90$ $v$-independent only & $0.7606$\\
$21$ $v$-dependent only & $0.9932$\\
all $111$ & $0.7328$\\
\hline
\end{tabular}
\end{center}
Alone the $v$-dependent family is worthless; jointly it contributes $2.8$ points, chiefly through
$\varphi_j Q_t(v)$ columns that cancel $Q$.

\paragraph{Is the $v$-dependence of $Q_t(v)=\langle Ah_t,Be_v\rangle$ useful?} No.
Replacing $Q_t(v)$ by its own mean $\bar q_t=\tau^{(1)}_t$ gives ratio $0.9954$ (slightly \emph{better}).
Held-out token $R^2$ for $f$: prefix-only $0.3369$; $+\,e_{y_t}$ $0.3377$; $+$ rank-32 bilinear
(exactly this head's class) $0.3384$. Under the sampling distribution,
$\E\,\Var_{v\sim p_t}\!\big(Q_t(v)\big)\,/\,\E\,a_t^2=0.017$ overall and $0.074$ on the top entropy decile,
while $57\%$ of tokens have $\max_v p_t(v)>0.9$: critical actions exist, but the policy is already
confident at them.

\paragraph{Ceiling.} The oracle $\Delta_t=f-\bar q_t$ (inadmissible: it reads the realised $f$) attains
$0.062$, so the dictionary is not the binding constraint. Empirically the ratio tracks
$1-0.77\,R^2_{\text{token}}$, while the best held-out $R^2$ for $f$ from the \emph{entire} sequence is
$0.554$. Reaching $0.50$ would need $R^2\approx0.65$. The residual is the black-box evaluation's own noise.
